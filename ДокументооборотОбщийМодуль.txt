Функция ПрочитатьУзел(Чтение)
	
	перем ИмяТекущегоТега;
	перем ЗначениеУзла;  
	перем ПредТип;
	СтруктураУзла = Новый Структура;
	ПредТип = "0";
	
	Пока Чтение.Прочитать() Цикл
		
		Если Чтение.ТипУзла = ТипУзлаXML.НачалоЭлемента Тогда
			//Начало вложенного тега
			ИмяТекущегоТега = Чтение.Имя;
			ЗначениеУзла = ПрочитатьУзел(Чтение);
			Если ЗначениеУзла = ПредТип тогда
				СтруктураУзла.Вставить(ИмяТекущегоТега, "");
			КонецЕсли;
			Если ИмяТекущегоТега = "Физлицо" Тогда
				СтруктураУзла.Вставить(ИмяТекущегоТега, ЗначениеУзла);
			КонецЕсли;
			
		ИначеЕсли Чтение.ТипУзла = ТипУзлаXML.Текст Тогда
			// Текстовое содержимое (значение между тегами)
			Возврат СокрЛП(Чтение.Значение);    
			
		ИначеЕсли Чтение.ТипУзла = ТипУзлаXML.КонецЭлемента Тогда   
			// Возвращаем заполненную структуру для родительского узла
			Если ИмяТекущегоТега <> Неопределено Тогда				
				СтруктураУзла.Вставить(ИмяТекущегоТега, ЗначениеУзла);
			Иначе
				Возврат ПредТип;  
			КонецЕсли;
		КонецЕсли;
		
	КонецЦикла;
	
	Возврат СтруктураУзла;
КонецФункции

Процедура ЗагрузитьИзмененияОбъекта(СтруктураДанных)
	//ЗаписьЖурналаРегистрации("обновляем данные", УровеньЖурналаРегистрации.Примечание, , , "Обработка сообщений");
	
	Попытка
		//проверяем есть ли снилс и инн
		Если НЕ ПустаяСтрока(СтруктураДанных.Физлицо.СтраховойНомерПФР) Тогда
			СНИЛС = СтруктураДанных.Физлицо.СтраховойНомерПФР;
		КонецЕсли;
		Если Не ПустаяСтрока(СтруктураДанных.Физлицо.ИНН) Тогда
			ИНН = СтруктураДанных.Физлицо.ИНН;
		КонецЕсли; 
		
		//ищем по снилсу
		Если Справочники.ФизическиеЛица.НайтиПоРеквизиту("СтраховойНомерПФР",СНИЛС).СтраховойНомерПФР <> "" 
			И НЕ ПустаяСтрока(СНИЛС) Тогда
			//Сообщить("Найден по СНИЛС");
			//получение объекта
			СправОбъект = Справочники.ФизическиеЛица.НайтиПоРеквизиту("СтраховойНомерПФР",СНИЛС);
			СправОбъект = СправОбъект.ПолучитьОбъект();
			
		//если не нашли по снилсу ищем по инн	
		ИначеЕсли Справочники.ФизическиеЛица.НайтиПоРеквизиту("ИНН",ИНН).ИНН <> "" 
			И НЕ ПустаяСтрока(ИНН) Тогда
			//Сообщить("Найден по ИНН"); 		
			//получение объекта
			СправОбъект = Справочники.ФизическиеЛица.НайтиПоРеквизиту("ИНН",ИНН).ИНН;
			СправОбъект = СправОбъект.ПолучитьОбъект();
			
		//если нет ни снилса ни инн в базе или сообщении - ищем по фио	
		ИначеЕсли Справочники.ФизическиеЛица.НайтиПоНаименованию(СтруктураДанных.Физлицо.Наименование).Наименование <> "" Тогда
			//Сообщить("Найден по ФИО"); 		
			//получение объекта
			СправОбъект = Справочники.ФизическиеЛица.НайтиПоНаименованию(СтруктураДанных.Физлицо.Наименование);
			СправОбъект = СправОбъект.ПолучитьОбъект();
			
		//если совсем нет - моздаем нового	
		Иначе
			//Сообщить("Создан");
			СправОбъект = Справочники.ФизическиеЛица.СоздатьЭлемент();
		КонецЕсли;
		//заполняем основные свойства
		ЗаполнитьЗначенияСвойств(СправОбъект, СтруктураДанных.Физлицо, "СтраховойНомерПФР, ИНН, ДатаРождения, Наименование, ПометкаУдаления");
		//пол заполняем отдельно, так как его надо достать из перечисления
		СправОбъект.Пол = Перечисления.ПолФизическогоЛица[СтруктураДанных.Физлицо.Пол];
		//очищаем контактную информацию
		СправОбъект.КонтактнаяИнформация.Очистить();		
		//заполняем болок контактной инфоормации, если свойства есть
		Если СтруктураДанных.Физлицо.Свойство("АдресМестаПроживанияФизическиеЛица") Тогда
			Адр = СправОбъект.КонтактнаяИнформация.Добавить();
			Адр.Вид = Справочники.ВидыКонтактнойИнформации.НайтиПоНаименованию("Домашний адрес");
			Адр.Тип = Перечисления.ТипыКонтактнойИнформации["Адрес"];
			
			Адр.Представление = СтруктураДанных.Физлицо.АдресМестаПроживанияФизическиеЛица;
		КонецЕсли;
		
		
		Если СтруктураДанных.Физлицо.Свойство("ТелефонДомашнийФизическиеЛица") Тогда
			Домтел = СправОбъект.КонтактнаяИнформация.Добавить();
			Домтел.Вид = Справочники.ВидыКонтактнойИнформации.НайтиПоНаименованию("Домашний телефон");
			Домтел.Тип = Перечисления.ТипыКонтактнойИнформации["Телефон"];
			
			Домтел.Представление = СтруктураДанных.Физлицо.ТелефонДомашнийФизическиеЛица;
		КонецЕсли;
		
		
		Если СтруктураДанных.Физлицо.Свойство("ТелефонМобильныйФизическиеЛица") Тогда
			Мобтел = СправОбъект.КонтактнаяИнформация.Добавить();
			Мобтел.Вид = Справочники.ВидыКонтактнойИнформации.НайтиПоНаименованию("Мобильный телефон");
			Мобтел.Тип = Перечисления.ТипыКонтактнойИнформации["Телефон"];
			Мобтел.НомерТелефона = СтруктураДанных.Физлицо.ТелефонМобильныйФизическиеЛица;
			
			Мобтел.Представление = СтруктураДанных.Физлицо.ТелефонМобильныйФизическиеЛица;
		КонецЕсли;
		Если СтруктураДанных.Физлицо.Свойство("EMailФизическиеЛица") Тогда
			Почт = СправОбъект.КонтактнаяИнформация.Добавить();
			Почт.Вид = Справочники.ВидыКонтактнойИнформации.НайтиПоНаименованию("Email");
			Почт.Тип = Перечисления.ТипыКонтактнойИнформации["АдресЭлектроннойПочты"];
			Почт.АдресЭП = СтруктураДанных.Физлицо.EMailФизическиеЛица;
			
			Почт.Представление = СтруктураДанных.Физлицо.EMailФизическиеЛица;
		КонецЕсли;
		
		СправОбъект.Записать();
	Исключение 
		// Запись ошибки в журнал регистрации с более подробной информацией
		ЗаписьЖурналаРегистрации("СервисыИнтегацииОбработка", УровеньЖурналаРегистрации.Ошибка, , , ОписаниеОшибки());
	КонецПопытки;
	
	//удаление обработанных сообщений позволит не обрабатывать их каждый раз, с новым сообщением
	СервисыИнтеграции.СервисДокументооборота.Основной_ФизЛицаИзУниВДок_ВДок.УдалитьСообщения(); 
КонецПроцедуры
