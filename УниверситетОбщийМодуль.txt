		Если Источник.ОбменДанными.Загрузка тогда
			Возврат;
		КонецЕсли;
		// Обрабатываем только объекты справочника "ФизическиеЛица"
		Если ТипЗнч(Источник) = Тип("СправочникОбъект.ФизическиеЛица") тогда

			ЗаписьXML.УстановитьСтроку();
			ЗаписьXML.ЗаписатьОбъявлениеXML();		
			
			// Формирование XML-данных 	
			ЗаписьXML.ЗаписатьНачалоЭлемента("Физлицо"); 
			
			// Запросы для получения дополнительных данных по физическому лицу:
			// - актуальное состояние из регистра "СостояниеСтудентов" (срез последних)
			// - контактная информация физлица

			СостояниеСтудента.Текст = 
			"ВЫБРАТЬ

			|	РегистрСведений.СостояниеСтудентов.СрезПоследних КАК СостояниеСтудентовСрезПоследних
			|ГДЕ
			|	СостояниеСтудентовСрезПоследних.ФизическоеЛицо.Ссылка = &Ссылка";
			СостояниеСтудента.УстановитьПараметр("Ссылка", Источник.Ссылка);	
			
			РезультатЗапроса = СостояниеСтудента.Выполнить();
			
			ВыборкаСостояние = РезультатЗапроса.Выбрать();
			
			Пока ВыборкаСостояние.Следующий() Цикл
				
				ЗаписьXML.ЗаписатьНачалоЭлемента("Комментарий"); 
				ЗаписьXML.ЗаписатьТекст(Источник.Комментарий + " " + ВыборкаСостояние.Состояние.Наименование);
				ЗаписьXML.ЗаписатьКонецЭлемента();
				
			КонецЦикла;
			
			// Записываем основные реквизиты физического лица в XML
			ЗаписьXML.ЗаписатьНачалоЭлемента("Ссылка");
			ЗаписьXML.ЗаписатьТекст(Строка(Источник.Ссылка));
			ЗаписьXML.ЗаписатьКонецЭлемента();
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("ПометкаУдаления");
			ЗаписьXML.ЗаписатьТекст(Строка(Источник.ПометкаУдаления));
			ЗаписьXML.ЗаписатьКонецЭлемента();		
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("Наименование");
			ЗаписьXML.ЗаписатьТекст(Источник.Наименование);		
			ЗаписьXML.ЗаписатьКонецЭлемента();
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("Пол"); 
			ЗаписьXML.ЗаписатьТекст(Источник.Пол.Наименование);
			ЗаписьXML.ЗаписатьКонецЭлемента();
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("ДатаРождения");
			ЗаписьXML.ЗаписатьТекст(Строка(Источник.ДатаРождения));
			ЗаписьXML.ЗаписатьКонецЭлемента();
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("ИНН");   
			ЗаписьXML.ЗаписатьТекст(Источник.ИНН);
			ЗаписьXML.ЗаписатьКонецЭлемента();
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("СтраховойНомерПФР"); 
			ЗаписьXML.ЗаписатьТекст(Источник.СНИЛС);
			ЗаписьXML.ЗаписатьКонецЭлемента();
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("ЭтоГруппа");   
			ЗаписьXML.ЗаписатьТекст(СТрока(Источник.ЭтоГруппа));
			ЗаписьXML.ЗаписатьКонецЭлемента();  
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("АдресМестаПроживанияФизическиеЛица");
			ЗаписьXML.ЗаписатьТекст(
			УправлениеКонтактнойИнформацией.КонтактнаяИнформацияОбъекта(
			Источник.Ссылка,
			УправлениеКонтактнойИнформацией.ВидКонтактнойИнформацииПоИмени("АдресМестаПроживанияФизическиеЛица"), 
			ТекущаяДата(),
			Истина
			)
			);			
			ЗаписьXML.ЗаписатьКонецЭлемента();
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("ТелефонДомашнийФизическиеЛица");
			ЗаписьXML.ЗаписатьТекст(
			УправлениеКонтактнойИнформацией.КонтактнаяИнформацияОбъекта(
			Источник.Ссылка,
			УправлениеКонтактнойИнформацией.ВидКонтактнойИнформацииПоИмени("ТелефонДомашнийФизическиеЛица"), 
			ТекущаяДата(),
			Истина
			)
			);			
			ЗаписьXML.ЗаписатьКонецЭлемента();
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("ТелефонМобильныйФизическиеЛица");
			ЗаписьXML.ЗаписатьТекст(
			УправлениеКонтактнойИнформацией.КонтактнаяИнформацияОбъекта(
			Источник.Ссылка,
			УправлениеКонтактнойИнформацией.ВидКонтактнойИнформацииПоИмени("ТелефонМобильныйФизическиеЛица"), 
			ТекущаяДата(),
			Истина
			)
			);			
			ЗаписьXML.ЗаписатьКонецЭлемента();
			
			ЗаписьXML.ЗаписатьНачалоЭлемента("EMailФизическиеЛица");
			ЗаписьXML.ЗаписатьТекст(
			УправлениеКонтактнойИнформацией.КонтактнаяИнформацияОбъекта(
			Источник.Ссылка,
			УправлениеКонтактнойИнформацией.ВидКонтактнойИнформацииПоИмени("EMailФизическиеЛица"), 
			ТекущаяДата(),
			Истина
			)
			);			
			ЗаписьXML.ЗаписатьКонецЭлемента();
			
			ЗаписьXML.ЗаписатьКонецЭлемента();// Конец "Физлицо"		
			
			// Завершаем формирование XML, получаем строку
			
			Стр = ЗаписьXML.Закрыть();
			Буфер = ПолучитьБуферДвоичныхДанныхИзСтроки(Стр);
			Тело.Записать(Буфер, 0, Буфер.Размер);
			Тело.Закрыть();
			// Передаём размер сообщения в параметры (может использоваться принимающей стороной)
			Сообщение.Параметры.Вставить("РазмерСообщения", Буфер.Размер);
			// Отправляем сформированное сообщение через канал "Основной_ФизЛицаИзУниВДок_ИзУни"
			СервисыИнтеграции.СервисУниверситета.Основной_ФизЛицаИзУниВДок_ИзУни.ОтправитьСообщение(Сообщение);
